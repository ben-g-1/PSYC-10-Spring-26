1. Central Limit Theorem (CLT)

The Central Limit Theorem states that the sampling distribution of the sample mean approaches a Normal distribution as the sample size (n) grows — regardless of the shape of the underlying population distribution.

Key takeaways:

  • The mean of the sampling distribution equals the population mean (μ)
  • The spread (standard error) shrinks as sample size n increases: SE = σ / √n
  • Even highly skewed or bimodal populations produce approximately Normal sampling distributions with enough samples

The tendency for our samples to create this type of normal distribution is what enables most of our other statistical tests.

How to use this app: Choose a population shape, set the sample size (n), then drag the “Number of samples” slider slowly from left to right to watch the sampling distribution build up. Try setting your sample size to 1, 10, and 100. Do you see much of a difference beyond that?

Notable milestones for a normally-distributed dataset: 1 sample (almost nothing), 10 samples (rough shape), 100 samples (clearly bell-shaped), 300 samples (well-converged).

However, note how the sample size changes how quickly the correct mean is estimated.

ui <- fluidPage(
  titlePanel("Central Limit Theorem"),
  sidebarLayout(
    sidebarPanel(
      selectInput("dist", "Population shape",
        choices = c("Normal" = "normal",
                    "Right-skewed" = "skewed",
                    "Uniform" = "uniform",

      sliderInput("n_sims", "Number of samples drawn", min = 1, max = 300, value = 1),
      hr(),

      plotOutput("cltPlot", height = "380px"),
      uiOutput("callout")
    )

server <- function(input, output) {

  population <- reactive({
    set.seed(110)
    switch(input$dist,
      normal  = rnorm(50, mean = 21, sd = 1.5),
      skewed  = c(rnorm(40, mean = 19, sd = 1), rnorm(10, mean = 27, sd = 2)),
      uniform = runif(200, min = 18, max = 26),
      bimodal = c(rnorm(100, mean = 18, sd = 1), rnorm(100, mean = 24, sd = 1))
    )
  })

  sample_means <- reactive({
    set.seed(110)
    pop <- population()
    replicate(input$n_sims, mean(sample(pop, input$n, replace = TRUE)))
  })

  output$cltPlot <- renderPlot({
    pop   <- population()
    means <- sample_means()
    n_sims <- input$n_sims
    true_mean <- mean(pop)

    par(mfrow = c(1, 2), mar = c(4, 4, 3, 1))

    # Panel 1: Population
    hist(pop,
         breaks = 20,
         col    = "skyblue",
         border = "white",
         main   = "Population Distribution",
         xlab   = "Value",
         xlim   = range(pop) + c(-1, 1))
    abline(v = true_mean, col = "red", lwd = 3)
    legend("topleft", legend = paste("μ =", round(true_mean, 2)),
           col = "red", lwd = 2, bty = "n", cex = 0.85)

    # Panel 2: Sampling distribution
    xlim_range <- range(c(means, pop)) + c(-1, 1)
    hist(means,
         breaks = max(5, min(30, n_sims %/% 5)),
         col    = "tomato",
         border = "white",
         main   = sprintf("Sampling Distribution\n(%d sample%s, n = %d)",
                          n_sims, if (n_sims == 1) "" else "s", input$n),
         xlab   = "Sample mean",
         xlim   = xlim_range)
    abline(v = true_mean,   col = "red",  lwd = 3)
    abline(v = mean(means), col = "blue", lwd = 3, lty = 2)
    legend("topleft",
           legend = c(paste("True μ =", round(true_mean, 2)),
                      paste("x̄ of means =", round(mean(means), 2))),
           col = c("red", "blue"), lwd = 2, lty = c(1, 2), bty = "n", cex = 0.85)
  })

  output$callout <- renderUI({
    means     <- sample_means()
    true_mean <- mean(population())
    n_sims    <- input$n_sims
    se_theory <- sd(population()) / sqrt(input$n)
    se_obs    <- sd(means)
    milestone <- if (n_sims == 1) {
      "Just 1 sample — almost no distribution to see yet."
    } else if (n_sims <= 10) {
      paste0("After ", n_sims, " samples: shape is very rough.")
    } else if (n_sims <= 100) {
      paste0("After ", n_sims, " samples: bell shape is emerging.")
    } else {
      paste0("After ", n_sims, " samples: well-converged to Normal.")
    }
    tags$div(
      style = "background:#f0f4f8; border-left:4px solid #2c7bb6; padding:10px 14px; margin-top:10px; border-radius:4px;",
      tags$b("Snapshot: "), milestone, tags$br(),
      sprintf("Running mean of sample means: %.3f (true μ = %.3f)", mean(means), true_mean), tags$br(),
      sprintf("Observed SE: %.3f | Theoretical SE (σ/√n): %.3f", se_obs, se_theory)
    )
  })
}

shinyApp(ui = ui, server = server)

2. Z-Test

A z-test is used when we know the population standard deviation (σ) — a rare but instructive scenario (e.g., when a large historical dataset gives us σ exactly).

\[z = \frac{\bar{x} - \mu_0}{\sigma / \sqrt{n}}\]

Because σ is known, we compare to the Standard Normal (not a t-distribution). The critical value is basically always ±1.96 for α = 0.05, two-tailed, regardless of n. This is directly in contrast to t-tests, where the critical values shrink as n incraeses.

How to use this app: Notice how the z-test and t-test give nearly identical results when n is large — the t-distribution approaches Normal.

ui <- fluidPage(
  titlePanel("Z-Test (Known σ)"),
  sidebarLayout(
    sidebarPanel(
      sliderInput("mu0",         "Hypothesized mean (μ₀)",  15, 25, 20, step = 0.5),
      sliderInput("sample_mean", "Observed sample mean (x̄)", 15, 25, 21, step = 0.5),
      sliderInput("sigma",       "Known population SD (σ)",  0.5, 8, 2,  step = 0.1),
      sliderInput("n",           "Sample size (n)",           5, 200, 30)
    ),
    mainPanel(
      plotOutput("zPlot", height = "300px"),
      uiOutput("zResults")
    )
  )

server <- function(input, output) {

  z_stat <- reactive({
    (input$sample_mean - input$mu0) / (input$sigma / sqrt(input$n))
  })

  p_val <- reactive({
    2 * pnorm(-abs(z_stat()))
  })

  output$zPlot <- renderPlot({
    z    <- z_stat()
    z_c  <- 1.96
    x    <- seq(-5, 5, length.out = 500)
    y    <- dnorm(x)

    par(mar = c(4, 4, 3, 1))
    plot(x, y, type = "l", lwd = 2,
         main = "Standard Normal Distribution",
         xlab = "z", ylab = "Density")
    x_lo <- x[x <= -z_c]; x_hi <- x[x >= z_c]
    polygon(c(min(x_lo), x_lo, -z_c), c(0, dnorm(x_lo), 0),
            col = rgb(0.5, 0.5, 0.5, 0.3), border = NA)
    polygon(c(min(x_hi), x_hi, z_c), c(0, dnorm(x_hi), 0),
            col = rgb(0.5, 0.5, 0.5, 0.3), border = NA)
    abline(v =  z,   col = "navy",   lwd = 2.5, lty = 2)
    abline(v =  z_c, col = "tomato", lwd = 1.5, lty = 3)
    abline(v = -z_c, col = "tomato", lwd = 1.5, lty = 3)
    legend("topleft",
           legend = c(sprintf("z = %.3f", z),
                      "±z_crit = ±1.96",
                      "Rejection region (α=0.05)"),
           col = c("navy", "tomato", rgb(0.5, 0.5, 0.5, 0.5)),
           lwd = c(2.5, 1.5, 8), lty = c(2, 3, 1), bty = "n", cex = 0.85)
  })

  output$zResults <- renderUI({
    z      <- z_stat()
    p      <- p_val()
    reject <- p < 0.05
    color  <- if (reject) "#c0392b" else "#27ae60"
    decision <- if (reject)
      sprintf("Reject H₀: μ₀ = %.1f (p < 0.05)", input$mu0)
    else
      sprintf("Fail to reject H₀: μ₀ = %.1f (p ≥ 0.05)", input$mu0)
    tags$div(
      style = paste0("background:#f9f9f9; border-left:4px solid ", color,
                     "; padding:10px 14px; margin-top:10px; border-radius:4px;"),
      tags$b("Results:"), tags$br(),
      sprintf("z = %.3f", z), tags$br(),
      sprintf("p-value = %.4f", p), tags$br(),
      tags$b(decision)
    )
  })
}

shinyApp(ui = ui, server = server)

3. One-Sample t-Test

A one-sample t-test asks: Is our sample mean significantly different from a hypothesized population mean (μ₀)?

We use a t-statistic (not z) because σ is unknown and must be estimated from the sample:

\[t = \frac{\bar{x} - \mu_0}{s / \sqrt{n}}, \quad df = n - 1\]

The p-value is the probability of observing a t-statistic this extreme (or more) if H₀ were true.

How to use this app: Adjust the sliders to see how moving μ₀ further from the observed mean, increasing n, or decreasing s all affect the t-statistic and p-value.

ui <- fluidPage(
  titlePanel("One-Sample t-Test"),
  sidebarLayout(
    sidebarPanel(
      sliderInput("mu0",         "Hypothesized mean (μ₀)",  15, 25, 20, step = 0.5),
      sliderInput("sample_mean", "Observed sample mean (x̄)", 15, 25, 21, step = 0.5),
      sliderInput("sample_sd",   "Sample SD (s)",            0.5, 5, 2,  step = 0.1),
      sliderInput("n",           "Sample size (n)",           5, 100, 30)
    ),
    mainPanel(
      plotOutput("tPlot", height = "320px"),
      uiOutput("results")
    )
  )
)

server <- function(input, output) {

  stats <- reactive({
    t_stat <- (input$sample_mean - input$mu0) / (input$sample_sd / sqrt(input$n))
    df     <- input$n - 1
    p_val  <- 2 * pt(-abs(t_stat), df)
    list(t = t_stat, df = df, p = p_val)
  })

  output$tPlot <- renderPlot({
    s    <- stats()
    df   <- s$df
    t_ob <- s$t
    t_crit <- qt(0.975, df)
    x    <- seq(-5, 5, length.out = 500)
    y    <- dt(x, df)

    par(mar = c(4, 4, 3, 1))
    plot(x, y, type = "l", lwd = 2,
         main = sprintf("t-Distribution (df = %d)", df),
         xlab = "t", ylab = "Density")

    # Shade rejection regions
    x_lo <- x[x <= -t_crit]; x_hi <- x[x >= t_crit]
    polygon(c(min(x_lo), x_lo, -t_crit), c(0, dt(x_lo, df), 0),
            col = rgb(0.5, 0.5, 0.5, 0.3), border = NA)
    polygon(c(min(x_hi), x_hi, t_crit), c(0, dt(x_hi, df), 0),
            col = rgb(0.5, 0.5, 0.5, 0.3), border = NA)

    abline(v = t_ob,    col = "navy",   lwd = 2.5, lty = 2)
    abline(v =  t_crit, col = "tomato", lwd = 1.5, lty = 3)
    abline(v = -t_crit, col = "tomato", lwd = 1.5, lty = 3)
    legend("topleft",
           legend = c(sprintf("t = %.3f", t_ob),
                      sprintf("±t_crit = ±%.3f", t_crit),
                      "Rejection region (α=0.05)"),
           col = c("navy", "tomato", rgb(0.5, 0.5, 0.5, 0.5)),
           lwd = c(2.5, 1.5, 8), lty = c(2, 3, 1), bty = "n", cex = 0.85)
  })

  output$results <- renderUI({
    s      <- stats()
    reject <- s$p < 0.05
    color  <- if (reject) "#c0392b" else "#27ae60"
    decision <- if (reject)
      sprintf("Reject H₀: μ₀ = %.1f (p < 0.05)", input$mu0)
    else
      sprintf("Fail to reject H₀: μ₀ = %.1f (p ≥ 0.05)", input$mu0)

    tags$div(
      style = paste0("background:#f9f9f9; border-left:4px solid ", color,
                     "; padding:10px 14px; margin-top:10px; border-radius:4px;"),
      tags$b("Results:"), tags$br(),
      sprintf("t(%d) = %.3f", s$df, s$t), tags$br(),
      sprintf("p-value = %.4f", s$p), tags$br(),
      tags$b(decision)
    )
  })
}

shinyApp(ui = ui, server = server)

4. Two-Sample t-Test

A two-sample t-test tests whether the means of two groups differ:

\[H_0: \mu_1 = \mu_2\]

We can use either independent or dependent tests based on what our two groups represent.

When to use which:

  • Independent samples t-test: Different subjects in each group (e.g., treatment vs. control with separate participants)
  • Paired t-test: Same subjects measured twice (e.g., pre-test vs. post-test, or same subject across two conditions)

4a. Independent samples

Two variants:

  • Pooled (equal variances assumed): combines variance estimates; uses n₁ + n₂ − 2 df
  • Welch (unequal variances): uses separate variance estimates with adjusted (Satterthwaite) df — generally safer, but not something you need to know for the midterm!. It might be instructive for you to see the difference between the two though, especially when setting the variances far apart from one another. The degrees of freedom for this version are adjusted.

CRITICAL: DON’T WORRY ABOUT THE WELCH TEST FOR ANYTHING OTHER THAN CURIOSITY (for now at least)

How to use this app: Slide Group 1 and Group 2 means apart to see when the test becomes significant. Toggle between pooled and Welch to compare.

server <- function(input, output) {

  samples <- reactive({
    set.seed(110)
    g1 <- rnorm(input$n1, input$mu1, input$sd1)
    g2 <- rnorm(input$n2, input$mu2, input$sd2)
    list(g1 = g1, g2 = g2)
  })

  test_result <- reactive({
    samp <- samples()
    t.test(samp$g1, samp$g2, var.equal = input$equal_var)
  })

  output$twoSamplePlot <- renderPlot({
    samp <- samples()
    res  <- test_result()
    t_ob <- res$statistic
    df   <- res$parameter
    t_crit <- qt(0.975, df)

    par(mfrow = c(1, 2), mar = c(4, 4, 3, 1))

    # Panel 1: overlapping histograms
    all_vals <- c(samp$g1, samp$g2)
    xlim_h   <- range(all_vals) + c(-1, 1)
    hist(samp$g1, breaks = 15, col = rgb(0.18, 0.53, 0.78, 0.6),
         border = "white", xlim = xlim_h,
         main = "Sample Distributions", xlab = "Value")
    hist(samp$g2, breaks = 15, col = rgb(0.82, 0.24, 0.24, 0.6),
         border = "white", add = TRUE)
    abline(v = mean(samp$g1), col = "blue", lwd = 2, lty = 2)
    abline(v = mean(samp$g2), col = "red",  lwd = 2, lty = 2)
    legend("topleft",
           legend = c(sprintf("G1: x̄=%.1f", mean(samp$g1)),
                      sprintf("G2: x̄=%.1f", mean(samp$g2))),
           fill = c(rgb(0.18, 0.53, 0.78, 0.6), rgb(0.82, 0.24, 0.24, 0.6)),
           bty = "n", cex = 0.85)

    # Panel 2: t-distribution
    x <- seq(max(-8, -abs(t_ob) - 3), max(8, abs(t_ob) + 3), length.out = 500)
    y <- dt(x, df)
    plot(x, y, type = "l", lwd = 2,
         main = sprintf("t-Distribution (df = %.1f)", df),
         xlab = "t", ylab = "Density")
    x_lo <- x[x <= -t_crit]; x_hi <- x[x >= t_crit]
    if (length(x_lo) > 1)
      polygon(c(min(x_lo), x_lo, -t_crit), c(0, dt(x_lo, df), 0),
              col = rgb(0.5, 0.5, 0.5, 0.3), border = NA)
    if (length(x_hi) > 1)
      polygon(c(min(x_hi), x_hi, t_crit), c(0, dt(x_hi, df), 0),
              col = rgb(0.5, 0.5, 0.5, 0.3), border = NA)
    abline(v =  t_ob,   col = "navy",   lwd = 2.5, lty = 2)
    abline(v =  t_crit, col = "tomato", lwd = 1.5, lty = 3)
    abline(v = -t_crit, col = "tomato", lwd = 1.5, lty = 3)
  })

  output$twoResults <- renderUI({
    res    <- test_result()
    t_ob   <- res$statistic
    df     <- res$parameter
    p_val  <- res$p.value
    reject <- p_val < 0.05
    color  <- if (reject) "#c0392b" else "#27ae60"
    method <- if (input$equal_var) "Pooled (equal variances)" else "Welch (unequal variances)"
    decision <- if (reject) "Reject H₀: μ₁ = μ₂" else "Fail to reject H₀: μ₁ = μ₂"
    tags$div(
      style = paste0("background:#f9f9f9; border-left:4px solid ", color,
                     "; padding:10px 14px; margin-top:10px; border-radius:4px;"),
      tags$b("Results (", method, "):"), tags$br(),
      sprintf("t(%.1f) = %.3f", df, t_ob), tags$br(),
      sprintf("p-value = %.4f", p_val), tags$br(),
      tags$b(decision)
    )
  })
}

4b. Paired Samples (Dependent t-Test)

A paired t-test is used when the same subjects are measured under two conditions (e.g., before/after treatment, left hand vs right hand, condition A vs condition B for the same person).

The key insight: by using the same subjects in both conditions, we can subtract out individual differences. This is analogous to the relationship between the between-groups ANOVA and within-groups (repeated measures) ANOVA in Section 5 — using the same subjects across conditions removes between-subject variability from the error term, dramatically increasing statistical power.

Instead of comparing two independent groups, we compute the difference score for each subject and test whether the mean difference is significantly different from zero:

\[t = \frac{\bar{d} - \mu_0}{s_d / \sqrt{n}}, \quad df = n - 1\]

where \(\bar{d}\) is the mean of the difference scores, \(s_d\) is the SD of the differences, and \(n\) is the number of pairs (not total observations).

How to use this app: Adjust the between-subject SD and within-subject (error) SD to see how the paired test gains power. When between-subject variability is high but within-subject error is low, the paired test dramatically outperforms the independent test on the same data.

ui <- fluidPage(
  titlePanel("Paired t-Test"),
  sidebarLayout(
    sidebarPanel(

      uiOutput("pairedResults")

server <- function(input, output) {

  paired_data <- reactive({
    set.seed(110)
    n <- input$n_pairs
    subj_fx <- rnorm(n, 0, input$sd_subj)  # individual baseline
    c1 <- subj_fx + rnorm(n, 0, input$sd_err)
    c2 <- subj_fx + input$mu_diff + rnorm(n, 0, input$sd_err)
    data.frame(subject = 1:n, c1 = c1, c2 = c2, diff = c2 - c1)
  })

  paired_test <- reactive({
    d <- paired_data()
    t.test(d$c2, d$c1, paired = TRUE)
  })

  indep_test <- reactive({
    d <- paired_data()
    t.test(d$c2, d$c1, paired = FALSE, var.equal = TRUE)
  })

  output$pairedPlot <- renderPlot({
    d <- paired_data()
    res <- paired_test()
    t_ob <- res$statistic
    df   <- res$parameter
    t_crit <- qt(0.975, df)

    par(mfrow = c(1, 2), mar = c(4, 4, 3, 1))

    # Panel 1: individual subject trajectories
    ylim_r <- range(c(d$c1, d$c2)) + c(-1, 1)
    plot(1:2, c(mean(d$c1), mean(d$c2)), type = "n",
         xlim = c(0.5, 2.5), ylim = ylim_r,
         xlab = "Condition", ylab = "Value",
         main = "Individual Subject Trajectories",
         xaxt = "n")
    axis(1, at = 1:2, labels = c("C1", "C2"))

    for (i in 1:nrow(d)) {
      lines(1:2, c(d$c1[i], d$c2[i]), col = rgb(0, 0, 0.8, 0.3), lwd = 1)
      points(1:2, c(d$c1[i], d$c2[i]), col = rgb(0, 0, 0.8, 0.4), pch = 16, cex = 0.8)
    }
    lines(1:2, c(mean(d$c1), mean(d$c2)), col = "navy", lwd = 3)
    points(1:2, c(mean(d$c1), mean(d$c2)), col = "navy", pch = 19, cex = 1.6)

    # Panel 2: histogram of difference scores
    hist(d$diff, breaks = 12, col = "tomato", border = "white",
         main = "Distribution of Difference Scores",
         xlab = "Difference (C2 - C1)")
    abline(v = 0, col = "gray40", lwd = 2, lty = 3)
    abline(v = mean(d$diff), col = "navy", lwd = 3, lty = 2)
    legend("topleft",
           legend = c("Zero (H₀)", sprintf("Mean diff = %.2f", mean(d$diff))),
           col = c("gray40", "navy"), lwd = 2, lty = c(3, 2), bty = "n", cex = 0.85)
  })

  output$pairedResults <- renderUI({
    res_p <- paired_test()
    res_i <- indep_test()
    t_p   <- res_p$statistic
    p_p   <- res_p$p.value
    t_i   <- res_i$statistic
    p_i   <- res_i$p.value
    reject <- p_p < 0.05
    color  <- if (reject) "#c0392b" else "#27ae60"
    decision <- if (reject) "Reject H₀: Mean difference ≠ 0"
                else "Fail to reject H₀: Mean difference = 0"

    tags$div(
      style = paste0("background:#f9f9f9; border-left:4px solid ", color,
                     "; padding:10px 14px; margin-top:10px; border-radius:4px;"),
      tags$b("Paired t-Test Results:"), tags$br(),
      sprintf("t(%.0f) = %.3f", res_p$parameter, t_p), tags$br(),
      sprintf("p-value = %.4f", p_p), tags$br(),
      tags$b(decision), tags$br(), tags$br(),
      tags$b("Compare to Independent Samples t-Test on same data:"), tags$br(),
      sprintf("t(%.0f) = %.3f, p = %.4f", res_i$parameter, t_i, p_i), tags$br(),
      tags$em("Notice how the paired test typically has a larger |t| and smaller p-value when between-subject variability is high.")
    )
  })
}

5. One-Way ANOVA

ANOVA (Analysis of Variance) tests whether 2 or more group means are equal. In practice, it’s usually used for 3 or more group means, but it’s important to understand that you can use it for comparing two groups. In this case, it actually gives you the same p-value as a pooled two-sample t-test! In other words, when performing an ANOVA for two groups: \[F = t^2\]

Remember that this is NOT the case when there are 3 or more groups, which is the usual time you’ll see ANOVA being used.

The null hypothesis is that there is no difference between any of the group means. If you get a significant p-value on these tests, you often will follow up with a t-test to identify which of the groups has the difference.

It partitions total variance into:

  • Between-groups (SS_B): variance due to group differences
  • Within-groups / Error (SS_W): variance within groups

The F-statistic = MS_Between / MS_Within. Large F indicates that the means differ more than chance.

\[F = \frac{MS_{Between}}{MS_{Within}} = \frac{SS_B / (k-1)}{SS_W / (N-k)}\]

5a. Between-Groups ANOVA

Independent groups — different subjects in each condition. This is roughly analogous to the independent-sample t-test. You assume that there is no underlying structure or shared variance between your groups in this case.

server <- function(input, output) {

  anova_data <- reactive({
    set.seed(110)
    k   <- input$k
    mus <- c(input$mu1g, input$mu2g, input$mu3g,
             if (k >= 4) input$mu4g else NULL,
             if (k >= 5) input$mu5g else NULL)[seq_len(k)]
    grp_list <- lapply(seq_len(k), function(i)
      data.frame(value = rnorm(input$n_g, mus[i], input$sd_g),
                 group = factor(paste0("G", i))))
    do.call(rbind, grp_list)
  })

  aov_result <- reactive({
    d <- anova_data()
    summary(aov(value ~ group, data = d))
  })

  output$anovaPlot <- renderPlot({
    d   <- anova_data()
    res <- aov_result()
    F_ob   <- res[[1]]$`F value`[1]
    df_bet <- res[[1]]$Df[1]
    df_wit <- res[[1]]$Df[2]
    F_crit <- qf(0.95, df_bet, df_wit)
    pal    <- c("skyblue", "tomato", "seagreen", "gold", "orchid")

    par(mfrow = c(1, 2), mar = c(4, 4, 3, 1))

    # Panel 1: boxplot of groups
    boxplot(value ~ group, data = d,
            col    = pal[seq_len(input$k)],
            border = "gray30",
            main   = "Group Distributions",
            xlab   = "Group", ylab   = "Value")
    stripchart(value ~ group, data = d,
               method = "jitter", add = TRUE, pch = 16,
               col = rgb(0, 0, 0, 0.3), vertical = TRUE)

    # Panel 2: F-distribution
    x_max <- max(F_crit * 2.5, F_ob * 1.3, 6)
    x <- seq(0, x_max, length.out = 500)
    y <- df(x, df_bet, df_wit)
    plot(x, y, type = "l", lwd = 2,
         main = sprintf("F(%d, %d) Distribution", df_bet, df_wit),
         xlab = "F", ylab = "Density")
    x_rej <- x[x >= F_crit]
    if (length(x_rej) > 1)
      polygon(c(F_crit, x_rej, max(x_rej)), c(0, df(x_rej, df_bet, df_wit), 0),
              col = rgb(0.5, 0.5, 0.5, 0.3), border = NA)
    abline(v = F_ob,   col = "navy",   lwd = 2.5, lty = 2)
    abline(v = F_crit, col = "tomato", lwd = 1.5, lty = 3)
    legend("topleft",
           legend = c(sprintf("F = %.3f", F_ob),
                      sprintf("F_crit = %.3f", F_crit)),
           col = c("navy", "tomato"), lwd = 2, lty = c(2, 3), bty = "n", cex = 0.85)
  })

  output$anovaTable <- renderTable({
    res <- aov_result()
    tbl <- res[[1]]
    data.frame(
      Source    = c("Between Groups", "Within Groups (Error)"),
      SS        = round(tbl$`Sum Sq`, 3),
      df        = tbl$Df,
      MS        = round(tbl$`Mean Sq`, 3),
      F         = c(round(tbl$`F value`[1], 3), NA),
      p         = c(round(tbl$`Pr(>F)`[1], 4), NA),
      check.names = FALSE
    )
  }, na = "", striped = TRUE, bordered = TRUE, hover = TRUE)

  output$anovaResult <- renderUI({
    res    <- aov_result()
    p_val  <- res[[1]]$`Pr(>F)`[1]
    reject <- p_val < 0.05
    color  <- if (reject) "#c0392b" else "#27ae60"
    decision <- if (reject) "Reject H₀: At least one group mean differs (p < 0.05)"
                else "Fail to reject H₀: No significant difference between group means (p ≥ 0.05)"
    tags$div(
      style = paste0("background:#f9f9f9; border-left:4px solid ", color,
                     "; padding:8px 14px; border-radius:4px;"),
      tags$b(decision)
    )
  })
}

5b. Within-Groups (Repeated Measures) ANOVA

The same subjects are measured across all conditions. Subject-to-subject variability is removed from the error term, dramatically increasing statistical power.

\[F_{RM} = \frac{MS_{Conditions}}{MS_{Conditions \times Subjects}}\]

The line plot below shows individual “subjects” across conditions. Parallel lines are good and indicate low within-subject error. This results in a high F.

REMEMBER: While t-scores are universally accepted as having clearly delineated conventional effect thresholds (d = 0.2 | 0.5 | 0.8), F tests do not have common effect sizes. Depending on your dataset, explaining 25% of the variance could be excellent! However, if you’re looking at a cancer screening test, it is likely far too low.

server <- function(input, output) {

  rm_data <- reactive({
    set.seed(110)
    k       <- input$k_rm
    n       <- input$n_subj
    subj_fx <- rnorm(n, 0, input$sd_subj)  # random subject intercepts
    cond_fx <- seq(0, input$mu_diff, length.out = k)

    rows <- expand.grid(subject = 1:n, condition = 1:k)
    rows$value <- subj_fx[rows$subject] + cond_fx[rows$condition] +
                  rnorm(nrow(rows), 0, input$sd_err)
    rows$subject   <- factor(rows$subject)
    rows$condition <- factor(rows$condition,
                             labels = paste0("C", seq_len(k)))
    rows
  })

  rm_aov <- reactive({
    d <- rm_data()
    summary(aov(value ~ condition + Error(subject/condition), data = d))
  })

  output$rmPlot <- renderPlot({
    d   <- rm_data()
    res <- rm_aov()

    # Extract F from within-subject stratum
    within_tbl <- res[["Error: subject:condition"]][[1]]
    F_ob   <- within_tbl$`F value`[1]
    df_cond <- within_tbl$Df[1]
    df_err  <- within_tbl$Df[2]
    F_crit  <- qf(0.95, df_cond, df_err)

    par(mfrow = c(1, 2), mar = c(4, 4, 3, 1))

    # Panel 1: individual trajectories
    conds   <- levels(d$condition)
    n_conds <- length(conds)
    cond_means <- tapply(d$value, d$condition, mean)
    ylim_r  <- range(d$value) + c(-0.5, 0.5)

    plot(1:n_conds, cond_means, type = "n",
         xlim = c(0.5, n_conds + 0.5), ylim = ylim_r,
         xlab = "Condition", ylab = "Value",
         main = "Individual Subject Trajectories",
         xaxt = "n")
    axis(1, at = 1:n_conds, labels = conds)

    for (subj in levels(d$subject)) {
      vals <- d$value[d$subject == subj]
      lines(1:n_conds, vals, col = rgb(0, 0, 0.8, 0.25), lwd = 1)
    }
    lines(1:n_conds, cond_means, col = "navy", lwd = 3)
    points(1:n_conds, cond_means, col = "navy", pch = 19, cex = 1.4)

    # Panel 2: F distribution
    x_max <- max(F_crit * 2.5, F_ob * 1.3, 6)
    x <- seq(0, x_max, length.out = 500)
    y <- df(x, df_cond, df_err)
    plot(x, y, type = "l", lwd = 2,
         main = sprintf("F(%d, %d) Distribution", df_cond, df_err),
         xlab = "F", ylab = "Density")
    x_rej <- x[x >= F_crit]
    if (length(x_rej) > 1)
      polygon(c(F_crit, x_rej, max(x_rej)), c(0, df(x_rej, df_cond, df_err), 0),
              col = rgb(0.5, 0.5, 0.5, 0.3), border = NA)
    abline(v = F_ob,   col = "navy",   lwd = 2.5, lty = 2)
    abline(v = F_crit, col = "tomato", lwd = 1.5, lty = 3)
    legend("topleft",
           legend = c(sprintf("F = %.3f", F_ob),
                      sprintf("F_crit = %.3f", F_crit)),
           col = c("navy", "tomato"), lwd = 2, lty = c(2, 3), bty = "n", cex = 0.85)
  })

  output$rmTable <- renderTable({
    res <- rm_aov()
    within_tbl <- res[["Error: subject:condition"]][[1]]
    data.frame(
      Source = c("Conditions (Between)", "Error (Conditions × Subjects)"),
      SS     = round(within_tbl$`Sum Sq`, 3),
      df     = within_tbl$Df,
      MS     = round(within_tbl$`Mean Sq`, 3),
      F      = c(round(within_tbl$`F value`[1], 3), NA),
      p      = c(round(within_tbl$`Pr(>F)`[1], 4), NA),
      check.names = FALSE
    )
  }, na = "", striped = TRUE, bordered = TRUE, hover = TRUE)

  output$rmResult <- renderUI({
    res        <- rm_aov()
    within_tbl <- res[["Error: subject:condition"]][[1]]
    p_val      <- within_tbl$`Pr(>F)`[1]
    reject     <- p_val < 0.05
    color      <- if (reject) "#c0392b" else "#27ae60"
    decision   <- if (reject)
      "Reject H₀: Condition means differ significantly (p < 0.05)"
    else
      "Fail to reject H₀: No significant effect of condition (p ≥ 0.05)"
    tags$div(
      style = paste0("background:#f9f9f9; border-left:4px solid ", color,
                     "; padding:8px 14px; border-radius:4px;"),
      tags$b(decision)
    )
  })
}
